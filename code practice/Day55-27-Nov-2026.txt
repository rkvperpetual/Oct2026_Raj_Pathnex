Jenkins Pipeline - Automatic Rollback
pipeline {
 agent any
 environment {
  INSTITUTE_NAME = "Pathnex"
  TEAM = "DevOps"
  ENV = "prod"
 }
 stages {
  stage('Deploy') {
   steps {
    script {
     try {
      sh 'echo "Deploying $INSTITUTE_NAME app in $ENV for $TEAM"'
      sh 'exit 1' //simulate failure
     } catch (Exception e) {
      echo "Deploying failed. Rolling back $INSTITUTE_NAME app in $ENV"     }
    }
   }
  }
 }
}
