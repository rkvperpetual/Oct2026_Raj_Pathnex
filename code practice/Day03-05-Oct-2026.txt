Ansible - Install Apache and Start services
 - name: Install and start Apache
 hosts: all
 become: yes
 tasks:
  - name:Install apache
  yum: 
   name: httpd
   state: present
  - name: Start apche
  service:
   name: httpd
   state: started
   enabled: yes

  name: pathnex-replicaset

  replicas: 3
  selector:
   matchLables:
    app: pathnex-app
  template:
   metadata:
   labels:
   app: pathnex-app
 spec:
  containers:
   - name: app
   image: nginx
   ports:
    - containerPort: 80
