Jenkins Pipeline - Cache Maven Dependencies
pipeline {
 agent any 
 environment {
  INSTITUTE_NAME = "Pathnex"
 }
 stages {
  stage('Checkout') {
   steps {
    git url: 'https://github.com/rkvperpetual/sentiment_app.git'
   }
  }
  stage('Build') {
   steps {
    sh '''
    if [ -d ".m2" ]; then
     echo "Using cached Dependencies"
    else
     mvn clean compile
    fi
    '''
   }
  }
 }
}
