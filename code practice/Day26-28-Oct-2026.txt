Ansible - Implementing SSH Key-based Authentication
- name: Implement SSH key Authentication
hosts: all
become: yes
tasks:
 - name: Generate SSH key pair
 command: ssh-keygen -t rsa -b 2048 -f /home/user/.ssh/id_rsa -N ""
 - name: Add SSH public key to authorized keys
 authorized_key:
  user: user
  state: present
  key: "{{ lookup('file', '/home/user/.ssh/id_rsa.pub')}}"

Kubernetes - Network Policies for Security
apiVersion: networking.k8s.io/v1
kind: NetworkPolicy
metadata:
 name: raj-web-policy
spec:
 podSelector:
  matchLabels:
   app: raj-web-app
 policyTypes:
  - Ingress
  - Egress
 ingress:
  - from:
   - podSelector:
    matchLabels:
     app: raj-api
 egress:
  - to:
   - podSelector:
    matchLabels:
     app: raj-database

Jenkinsfile - Secure the pipeline with Secrets
pipeline {
 agent any
 environment {
  AWS_ACCESS_KEY_ID = credentials('aws-access-key')
  AWS_SECRET_ACCESS_KEY = credentials('aws-secrets-key')
 }
 stages {
  stage('Build') {
   steps {
    echo 'Building Docker Image'
    docker.build('raj-web-app')
   }
  }
  stage('Deloy') {
   steps {
    echo 'Deploying to AWS EC2'
    sh 'aws ec2 run-instances --image-id ami-0abcd1234 --count 1 --instance-type t3.large'
   }
  }
 }
}

GitLab CI/CD - Use GitLab Secrets for Secure Deployment

 - build
 - deploy

  - echo "Deploying with Secure Credentials"
  - aws ec2 run-instances --image-id ami-0dhdg3423 --count 1 --instance-type c7i-flex.2xlarge --access-key $AWS_ACCESS_KEY --secret-key $AWS_SECRET_KEY
