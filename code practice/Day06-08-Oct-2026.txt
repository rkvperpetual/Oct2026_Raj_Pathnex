Ansible - Install Docker
 - name: Install Docker
 hosts: all
 become: yes
 tasks: 
  - name: Install Docker
  yum:
   name: docker
   state: present
  - name: Start Docker
  service:
   name: docker
   state: started
   enabled: yes

Kubernetes - Pod with Volume Mount
 apiVersion: v1
 kind: pod
 metadata:
  name: pathnex-pod
 spec:
  containers:
   - name: web
   image: nginx
   volumeMounts:
    - mountpath: /usr/share/nginx/html
    name: html-volume
  volumes:
   - name: html-volume
   hostPath:
    path: /data
    type: Directory

Shell Script - check for running services
#!/bin/bash
 services=("nginx" "docker" "httpd")
 for service in "${services[@]}"
 do
  if systemctl is-active --quiet $sevice; then
   echo "Service is running"
  else
   echo "$service is not running"
  fi
 done
