Terraform - EC2 with security group and ebs volume
resource "aws_security_group" "allow_ssh_http" {
 name = "allow_ssh_http"
 description = "allow ssh and http access"
 ingress {
  from_port = 22
  to_port = 22
  protocol = "tcp"
  cidr_blocks = ["0.0.0.0/0"]
 }
 ingress {
  from_port = 80
  to_port = 80
  protocol = "tcp"
  cidr_blocks = ["0.0.0.0/0"]
 }
}

resource "aws_instance" "pathnex_ec2" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "t3.medium"
  security_groups = [aws_security_group.allow_ssh_http.name]

  ebs_block_device {
    device_name = "/dev/sdh"
    volume_size = 30
  }
}

Jenkinsfile - docker build and push
pipeline {
 agent any 
 stages {
  stage('Build Docker Image') {
   steps {
    script {
     docker.build('pathnex-app')
    }
   }
 }
 stage('Push Docker Image') {
  steps {
   script {
    docker.withRegistry('https://docker.io', 'docker-credentials') {
     docker.image('pathnex-app').push('latest')
    }
   }
  }
 }
}

GitLab CI/CD - build & push docker image
stages:
 - build
 - push 

build:
 stage: build
 script:
  - docker build -t pathnex-app .
  
push:
 stage: push
 script:
  - docker login -u "$CI_REGISTRY_USER" -p "$CI_RESGISTRY_PASSWORD"
  - docker push pathnex-app
