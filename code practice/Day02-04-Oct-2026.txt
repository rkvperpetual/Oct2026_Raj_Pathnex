Ansible - Install & Enable Nginx
 - name: Install and start ngnix on Pathnex
 hosts: all
 became: yes
 
 tasks:
  - name: Install nginx
    yum:
     name: nginx
     state: present
 
 - name: Enable nginx
 service: 
  name: nginx
  state: started
  enabled: yes
