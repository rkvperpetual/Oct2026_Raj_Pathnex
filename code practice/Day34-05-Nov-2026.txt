Jenkins Pipeline
pipeline {
 agent any 
 stages {
  stage('checkout') {
   steps {
    git branch main https://github.com/Pathnex/sample-java-app.git
   }
  }
  stage('Build') {
   steps {
    sh 'mvn clean compile || {echo "Compile failed"; exit 1;}'
   }
  }
  stage('Test') {
   steps {
    sh 'mvn test || {echo "Test failed"; exit 1}'
   }
  }
 }
}
