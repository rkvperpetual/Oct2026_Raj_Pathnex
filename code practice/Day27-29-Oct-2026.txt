Ansible - Deploy Serverless Lambda Functions
- name: Deploy Serverless Lambda Functions
hosts: localhost
tasks:
 - name: Create Lambda Function
 command: aws lambda create-function --function-name pathnex-lambda --runtime nodejs14.x --role arn:aws:iam::5454524242:role/lambda-role --handler index.handler --zip-file fileb://lambda.zip

Kubernetes - Deploy Serverless Using Kubeless
apiVersion: kubeless.io/v1beta1
kind: Function
metadata: 
 name: raj-lambda
 namespace: default
spec:
 runtime: nodejs14
 handler: index.handler
 function: |
  module.exports = function(event, context) {
   context.succeed('Hello Raj');
  };
 
Jenkinsfile - Lambda Function Deployment Pipeline
pipeline {
 agent any
 stages {
  stage('Build') {
   steps {
    echo 'Building Lambda Function package'
    sh 'zip -r lambda.zip lambda/'
   }
  }
  stage('Deploy') {
   steps {
    script {
     sh 'aws lambda update-function-code --function-name raj-m=lambda --zip-file fileb://lambda.zip'
    }
   }
  }
 }
}
